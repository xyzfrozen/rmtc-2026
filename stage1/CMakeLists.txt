cmake_minimum_required(VERSION 3.16)
project(rmtc_stage1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)

add_library(collector_core src/collector.cpp)
target_include_directories(collector_core PUBLIC include)
target_link_libraries(collector_core PUBLIC PkgConfig::JSONCPP)
target_compile_options(collector_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(collector_node src/node.cpp)
target_link_libraries(collector_node collector_core)
ament_target_dependencies(collector_node rclcpp std_msgs)
target_compile_options(collector_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(collector_replay src/replay.cpp)
target_link_libraries(collector_replay collector_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(collector_tests tests/core_tests.cpp)
  target_link_libraries(collector_tests collector_core)
  add_test(NAME collector_contract_and_strategy COMMAND collector_tests)
endif()
install(TARGETS collector_node collector_replay DESTINATION lib/${PROJECT_NAME})
ament_package()
