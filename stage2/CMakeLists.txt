cmake_minimum_required(VERSION 3.16)
project(rmtc_stage2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
find_package(Threads REQUIRED)
add_library(battle_core src/protocol.cpp src/events.cpp src/combat.cpp src/bot.cpp deps/stage1/src/collector.cpp)
target_include_directories(battle_core PUBLIC include deps/stage1/include)
target_link_libraries(battle_core PUBLIC PkgConfig::JSONCPP)
target_compile_options(battle_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(battle_ai src/gateway.cpp)
target_include_directories(battle_ai PUBLIC include)
target_link_libraries(battle_ai PUBLIC battle_core PkgConfig::CURL Threads::Threads)
target_compile_options(battle_ai PRIVATE -Wall -Wextra -Wpedantic)
add_executable(battle_node src/node.cpp)
target_link_libraries(battle_node battle_ai)
ament_target_dependencies(battle_node rclcpp std_msgs)
include(CTest)
if(BUILD_TESTING)
 add_executable(battle_tests tests/core_tests.cpp)
 target_link_libraries(battle_tests battle_core)
 add_test(NAME battle_contract_and_strategy COMMAND battle_tests)
 add_executable(gateway_probe tests/gateway_probe.cpp)
 target_link_libraries(gateway_probe battle_ai)
 add_executable(fusion_tests tests/fusion_tests.cpp)
 target_link_libraries(fusion_tests battle_core)
 add_test(NAME fusion_navigation_and_inputs COMMAND fusion_tests)
 add_executable(battle_replay tests/replay.cpp)
 target_link_libraries(battle_replay battle_core)
endif()
install(TARGETS battle_node DESTINATION lib/${PROJECT_NAME})
ament_package()
